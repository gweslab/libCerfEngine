function(cerfengine_test name)
    add_executable(${name} ${name}.cpp test_host.cpp)
    target_link_libraries(${name} PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,cerfengine::cerfengine>")
    cerfengine_strict_warnings(${name})
    if(cerfengine_backend_x86_win32)
        target_compile_definitions(${name} PRIVATE CERFENGINE_TEST_BACKEND=1)
    endif()
    add_test(NAME ${name} COMMAND ${name})
endfunction()
