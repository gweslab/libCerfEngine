cmake_minimum_required(VERSION 3.25)
project(cerfengine VERSION 1.0.0 LANGUAGES CXX)

set(CERFENGINE_SERVICELOCATOR_DIR "${CMAKE_CURRENT_SOURCE_DIR}/libservicelocator"
    CACHE PATH "The libservicelocator checkout that the engine builds against")
if(NOT TARGET servicelocator::servicelocator)
    add_subdirectory("${CERFENGINE_SERVICELOCATOR_DIR}" servicelocator EXCLUDE_FROM_ALL)
endif()

if(MSVC)
    foreach(cerfengine_flags CMAKE_CXX_FLAGS_RELEASE CMAKE_CXX_FLAGS_RELWITHDEBINFO)
        string(REGEX REPLACE "/Ob[0-3]" "" ${cerfengine_flags} "${${cerfengine_flags}}")
        string(APPEND ${cerfengine_flags} " /Ob3")
    endforeach()
endif()

if(MSVC)
    string(TOLOWER "${CMAKE_CXX_COMPILER_ARCHITECTURE_ID}" cerfengine_cpu)
else()
    string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" cerfengine_cpu)
endif()
if(cerfengine_cpu MATCHES "^(x86|x64|x86_64|amd64|i[3-6]86)$")
    set(cerfengine_x86_translator ON)
else()
    set(cerfengine_x86_translator OFF)
endif()
if(MSVC AND cerfengine_cpu STREQUAL "x86")
    set(cerfengine_backend_x86_win32 ON)
else()
    set(cerfengine_backend_x86_win32 OFF)
endif()
if(NOT MSVC AND CMAKE_SYSTEM_NAME STREQUAL "Linux" AND cerfengine_cpu MATCHES "^(x86_64|amd64)$")
    set(cerfengine_backend_x86_64_sysv ON)
else()
    set(cerfengine_backend_x86_64_sysv OFF)
endif()
if(cerfengine_backend_x86_win32 OR cerfengine_backend_x86_64_sysv)
    set(cerfengine_execution_backend ON)
else()
    set(cerfengine_execution_backend OFF)
endif()

set(cerfengine_sources src/engine.cpp)
if(cerfengine_backend_x86_win32)
    list(APPEND cerfengine_sources src/host_platform_win32.cpp)
elseif(cerfengine_backend_x86_64_sysv)
    list(APPEND cerfengine_sources src/host_platform_linux.cpp)
else()
    list(APPEND cerfengine_sources src/host_platform_no_backend.cpp)
endif()
if(cerfengine_x86_translator)
    file(GLOB_RECURSE cerfengine_translator_sources CONFIGURE_DEPENDS
         "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
    list(REMOVE_ITEM cerfengine_translator_sources
         "${CMAKE_CURRENT_SOURCE_DIR}/src/engine.cpp"
         "${CMAKE_CURRENT_SOURCE_DIR}/src/host_platform_win32.cpp"
         "${CMAKE_CURRENT_SOURCE_DIR}/src/host_platform_linux.cpp"
         "${CMAKE_CURRENT_SOURCE_DIR}/src/host_platform_no_backend.cpp")
    list(APPEND cerfengine_sources ${cerfengine_translator_sources})
endif()

add_library(cerfengine STATIC ${cerfengine_sources})
add_library(cerfengine::cerfengine ALIAS cerfengine)
if(cerfengine_backend_x86_win32)
    set_target_properties(cerfengine PROPERTIES
        INTERPROCEDURAL_OPTIMIZATION_RELEASE ON
        INTERPROCEDURAL_OPTIMIZATION_RELWITHDEBINFO ON)
    target_link_options(cerfengine INTERFACE $<$<CONFIG:Release,RelWithDebInfo>:/LTCG>)
endif()
target_include_directories(cerfengine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(cerfengine PUBLIC cxx_std_20)
target_link_libraries(cerfengine PUBLIC servicelocator::servicelocator)
if(cerfengine_x86_translator)
    target_compile_definitions(cerfengine PRIVATE CERFENGINE_X86_TRANSLATOR=1)
endif()
if(cerfengine_backend_x86_win32)
    target_compile_definitions(cerfengine PRIVATE CERFENGINE_BACKEND_X86_WIN32=1)
endif()
if(cerfengine_backend_x86_64_sysv)
    target_compile_definitions(cerfengine PRIVATE CERFENGINE_BACKEND_X86_64_SYSV=1)
endif()
if(cerfengine_execution_backend)
    target_compile_definitions(cerfengine PRIVATE CERFENGINE_EXECUTION_BACKEND=1)
endif()

function(cerfengine_strict_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
            target_compile_options(${target} PRIVATE -Wno-maybe-uninitialized)
        endif()
    endif()
endfunction()
cerfengine_strict_warnings(cerfengine)

if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    set(cerfengine_is_top_level ON)
else()
    set(cerfengine_is_top_level OFF)
endif()
option(CERFENGINE_BUILD_TESTS "Build the libCerfEngine tests" ${cerfengine_is_top_level})

if(CERFENGINE_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
